cmake_minimum_required(VERSION 3.8)
project(d1h_robot_bridge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
add_executable(fsm_filter src/fsm_filter.cpp)
target_compile_features(fsm_filter PUBLIC cxx_std_17)
target_compile_options(fsm_filter PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(fsm_filter rclcpp std_msgs controller_manager_msgs)
install(TARGETS fsm_filter DESTINATION lib/${PROJECT_NAME})
ament_package()
